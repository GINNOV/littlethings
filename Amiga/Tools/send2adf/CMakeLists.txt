cmake_minimum_required(VERSION 3.24)
project(send2adf VERSION 1.5.0 LANGUAGES C)

option(SEND2ADF_TESTING "Permit test-only dependency manifest override" OFF)
option(ADFLIB_CANARY "Use a complete test-only canary identity override" OFF)
option(SEND2ADF_PRODUCTION_BUILD "Enforce install/package-safe dependency settings" OFF)
set(SEND2ADF_SANITIZERS "" CACHE STRING "Comma-separated compiler sanitizers for tests")

if(SEND2ADF_PRODUCTION_BUILD AND SEND2ADF_TESTING)
    message(FATAL_ERROR "production_testing_mode_rejected")
endif()
if(SEND2ADF_PRODUCTION_BUILD AND ADFLIB_CANARY)
    message(FATAL_ERROR "production_canary_mode_rejected")
endif()

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

include(cmake/ResolveADFlib.cmake)
send2adf_resolve_adflib()

string(TIMESTAMP SEND2ADF_BUILD_DATE "%Y%m%d" UTC)
string(REPLACE "\"" "\\\"" SEND2ADF_ADFLIB_IDENTITY_ESCAPED "${SEND2ADF_ADFLIB_IDENTITY}")
string(REPLACE "\"" "\\\"" SEND2ADF_ADFLIB_TRANSPORT_ESCAPED "${SEND2ADF_ADFLIB_TRANSPORT}")
configure_file(cmake/send2adf_version.h.in generated/send2adf_version.h @ONLY)
configure_file(cmake/InstallGuard.cmake.in generated/InstallGuard.cmake @ONLY)

add_executable(send2adf send2adf.c send2adf_runtime.c)
target_include_directories(send2adf PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(send2adf PRIVATE adf)
if(BUILD_TESTING)
    target_compile_definitions(send2adf PRIVATE SEND2ADF_TEST_HOOKS=1)
endif()

if(SEND2ADF_SANITIZERS)
    if(MSVC)
        message(FATAL_ERROR "send2adf_sanitizers_require_clang_or_gcc")
    endif()
    set(_send2adf_sanitize_flag "-fsanitize=${SEND2ADF_SANITIZERS}")
    target_compile_options(send2adf PRIVATE "${_send2adf_sanitize_flag}" -fno-omit-frame-pointer)
    target_link_options(send2adf PRIVATE "${_send2adf_sanitize_flag}")
    if(SEND2ADF_SANITIZERS MATCHES "address")
        target_compile_options(adf PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    endif()
endif()
if(MSVC)
    target_compile_options(send2adf PRIVATE /W4)
else()
    target_compile_options(send2adf PRIVATE -Wall -Wextra -Wpedantic)
endif()

install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/generated/InstallGuard.cmake")
install(TARGETS send2adf RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
